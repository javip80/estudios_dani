"ES_EN" : "EN_ES"; } // ===================================== // GENERACIÓN DE PREGUNTAS Y LÓGICA // ===================================== function buildMainTest(){ // Proporciones: 10 de M6, 25 del resto, 5 básicas let obligatoriasM6 = shuffle([...vocabM6]).slice(0, 10); let obligatoriasM1to5 = shuffle([...vocabM1to5]).slice(0, 25); let obligatoriasBasics = shuffle([...vocabBasics]).slice(0, 5); let seleccionadas = shuffle([...obligatoriasM6, ...obligatoriasM1to5, ...obligatoriasBasics]); currentQuestions = seleccionadas.map(item => ({ item, direction: getRandomDirection() })); reviewMode = false; } function buildReviewTest(){ currentQuestions = shuffle(mistakes).map(item => ({ item, direction: getRandomDirection() })); reviewMode = true; } function getWrongOptions(correctItem, direction){ const field = direction === "ES_EN" ? "en" : "es"; // Primero, busca opciones falsas dentro de su misma categoría para que sea más difícil let sameCategory = fullVocabulary.filter(v => v.category === correctItem.category && v[field] !== correctItem[field] ); let pool = sameCategory; // Si no hay suficientes palabras en esa categoría, coge del resto del vocabulario if(pool.length < 5){ pool = fullVocabulary.filter(v => v[field] !== correctItem[field]); } return shuffle([...new Set(pool.map(v => v[field]))]).slice(0,5); } // ===================================== // FLUJO DEL TEST // ===================================== function startMainTest(){ currentIndex = 0; correctAnswers = 0; wrongAnswers = 0; mistakes = []; buildMainTest(); questionTotal.textContent = MAIN_TEST_QUESTIONS; showQuizScreen(); renderQuestion(); } function startReviewMode(){ currentIndex = 0; correctAnswers = 0; wrongAnswers = 0; buildReviewTest(); questionTotal.textContent = currentQuestions.length; showQuizScreen(); renderQuestion(); } function renderQuestion(){ nextBtn.classList.add("hidden"); updateStats(); const current = currentQuestions[currentIndex]; const item = current.item; const direction = current.direction; const correct = direction === "ES_EN" ? item.en : item.es; const question = direction === "ES_EN" ? `How do you say "${item.es}" in English?` : `What does "${item.en}" mean?`; questionText.textContent = question; questionNumber.textContent = currentIndex + 1; progressBar.style.width = `${((currentIndex) / currentQuestions.length) * 100}%`; const wrongOptions = getWrongOptions(item, direction); // Ahora options tiene 6 elementos (1 correcta + 5 incorrectas generadas por slice(0,5)) const options = shuffle([correct, ...wrongOptions]); optionsContainer.innerHTML = ""; options.forEach(option => { const btn = document.createElement("button"); btn.className = "option-btn"; btn.textContent = option; btn.addEventListener("click", () => { const buttons = document.querySelectorAll(".option-btn"); buttons.forEach(b => { b.disabled = true; if(b.textContent === correct){ b.classList.add("correct-answer"); } }); if(option === correct){ correctAnswers++; }else{ wrongAnswers++; btn.classList.add("wrong-answer"); // --- LÓGICA CORREGIDA PARA LA PISTA --- const field = direction === "ES_EN" ? "en" : "es"; const clickedItem = fullVocabulary.find(v => v[field] === option); if(clickedItem){ const translation = direction === "ES_EN" ? clickedItem.es : clickedItem.en; const hint = document.createElement("span"); hint.className = "hint"; hint.textContent = `(Significa: ${translation})`; btn.appendChild(hint); } // ------------------------------------- if(!mistakes.includes(item)){ mistakes.push(item); } } updateStats(); nextBtn.classList.remove("hidden"); }); optionsContainer.appendChild(btn); }); } function nextQuestion(){ currentIndex++; if(currentIndex < currentQuestions.length){ renderQuestion(); }else{ showResults(); } } function showResults(){ quizScreen.classList.add("hidden"); resultScreen.classList.remove("hidden"); const total = currentQuestions.length; const percentage = Math.round((correctAnswers / total) * 100); finalScore.textContent = `${correctAnswers} / ${total}`; finalPercentage.textContent = `${percentage}%`; if(percentage >= PASS_PERCENTAGE){ resultTitle.textContent = "🏆 Excellent Work!"; resultTitle.className = "result-title success"; }else{ resultTitle.textContent = "❌ Keep Practising!"; resultTitle.className = "result-title fail"; } if(mistakes.length > 0 && !reviewMode){ mistakesSection.classList.remove("hidden"); mistakesList.innerHTML = ""; mistakes.forEach(word => { const li = document.createElement("li"); li.textContent = `${word.es} → ${word.en}`; mistakesList.appendChild(li); }); }else{ mistakesSection.classList.add("hidden"); } } function updateStats(){ correctCount.textContent = correctAnswers; wrongCount.textContent = wrongAnswers; modeBadge.textContent = reviewMode ? "Review Mode" : "Main Test"; } function showQuizScreen(){ resultScreen.classList.add("hidden"); quizScreen.classList.remove("hidden"); } // ===================================== // EVENTOS // ===================================== nextBtn.addEventListener("click", nextQuestion); restartBtn.addEventListener("click", startMainTest); reviewBtn.addEventListener("click", startReviewMode); // INICIAR TEST startMainTest();
